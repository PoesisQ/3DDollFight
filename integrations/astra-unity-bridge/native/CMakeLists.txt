cmake_minimum_required(VERSION 3.15)
project(AstraUnityBridge LANGUAGES CXX)
set(ASTRA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../lab-env/astra/sdk" CACHE PATH "Astra SDK directory")
add_executable(AstraUnityBridge main.cpp)
target_compile_features(AstraUnityBridge PRIVATE cxx_std_17)
target_compile_definitions(AstraUnityBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(AstraUnityBridge PRIVATE "${ASTRA_ROOT}/include")
target_link_libraries(AstraUnityBridge PRIVATE "${ASTRA_ROOT}/lib/astra.lib" "${ASTRA_ROOT}/lib/astra_core.lib" ws2_32)
target_compile_options(AstraUnityBridge PRIVATE /W4 /EHsc)
set_target_properties(AstraUnityBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
